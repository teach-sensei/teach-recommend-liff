      sh.setFrozenRows(1);
      console.log('作成: ' + pair[0]);
    } else {
      console.log('既存: ' + pair[0]);
    }
  });
}

// ========================================
// userId → 保護者ID（所有検証の共通処理）
// ========================================
function planResolveParent_(userId) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('紐付けマスタ');
  if (!sh) return null;
  var d = sh.getDataRange().getValues();
  var h = d[0];
  var uc = h.indexOf('LINE_userId');
  var pc = h.indexOf('保護者ID');
  var nc = h.indexOf('保護者の氏名');
  var oc = h.indexOf('配信オプトイン');
  for (var i = 1; i < d.length; i++) {
    if (d[i][uc] === userId) {
      return { parentId: d[i][pc], parentName: nc >= 0 ? d[i][nc] : '', optIn: oc >= 0 ? d[i][oc] : true };
    }
  }
  return null;
}

// plan_get：初期表示データ
// ========================================
function planGet(data) {
  var userId = data && data.userId;
  if (!userId) return { success: false, code: 'INVALID_INPUT' };

  var link = planResolveParent_(userId);
  if (!link) return { success: false, code: 'NOT_LINKED' };
  var parentId = link.parentId;

  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1) students_lite から自分の子を集める
  var stSh = ss.getSheetByName('students_lite');
  var sd = stSh.getDataRange().getValues();
  var sh0 = sd[0];
  var sSid = sh0.indexOf('生徒ID'), sPid = sh0.indexOf('保護者ID'), sNick = sh0.indexOf('ニックネーム');
  var mine = [];
  for (var i = 1; i < sd.length; i++) {
    if (String(sd[i][sPid]) === String(parentId)) {
      mine.push({ studentId: String(sd[i][sSid]), studentName: sd[i][sNick] || '' });
    }
  }
  if (!mine.length) return { success: false, code: 'NOT_FOUND' };

  // 2) report_source からお悩み・ペース
  var rsMap = {};
  var rsSh = ss.getSheetByName('report_source');
  if (rsSh) {
    var rd = rsSh.getDataRange().getValues();
    var rh = rd[0];
    var rSid = rh.indexOf('生徒ID'), rNay = rh.indexOf('主お悩み'), r30 = rh.indexOf('直近30日受講数_済');
    for (var j = 1; j < rd.length; j++) {
      rsMap[String(rd[j][rSid])] = { nayami: rd[j][rNay], r30: Number(rd[j][r30]) || 0 };
    }
  }

  // 3) lessons_lite から直近の受講時刻（受けやすい時間帯の初期値に使う）
  var lastTimeMap = {};
  var lsSh = ss.getSheetByName('lessons_lite');
  if (lsSh) {
    var ld = lsSh.getDataRange().getValues();
    var lh = ld[0];
    var lSid = lh.indexOf('生徒ID'), lEnd = lh.indexOf('授業の終了日時'), lConf = lh.indexOf('確定');
    var now = new Date();
    for (var k = 1; k < ld.length; k++) {
      var end = ld[k][lEnd];
      if (!end || typeof end.getMonth !== 'function') continue;   // ダックタイピングでDate判定
      if (end > now) continue;                                     // 未来の予約は除く
      var cf = String(ld[k][lConf]);
      if (cf !== '1' && cf !== '確定') continue;
      var sid = String(ld[k][lSid]);
      if (!lastTimeMap[sid] || end > lastTimeMap[sid].at) {
        lastTimeMap[sid] = { at: end };
      }
    }
  }

  // 4) 既に登録済みの未来の予定
  var existing = {};
  var pSh = ss.getSheetByName(PLAN_SHEET);
  if (pSh && pSh.getLastRow() > 1) {
    var pd = pSh.getDataRange().getValues();
    var ph = pd[0];
    var pSid = ph.indexOf('生徒ID'), pDate = ph.indexOf('予定日'), pTime = ph.indexOf('予定時刻'),
        pCont = ph.indexOf('やること'), pStat = ph.indexOf('状態');
    var todayStr = planFmtDate_(new Date());
    for (var m = 1; m < pd.length; m++) {
      if (String(pd[m][pStat]) !== '予定') continue;
      var ds = planToDateStr_(pd[m][pDate]);
      if (!ds || ds < todayStr) continue;
      var s2 = String(pd[m][pSid]);
      if (!existing[s2]) existing[s2] = [];
      existing[s2].push({ date: ds, time: planToTimeStr_(pd[m][pTime]), content: pd[m][pCont] || '' });
    }
  }

  // 5) 生徒ごとに組み立て
  var students = mine.map(function (s) {
    var rs = rsMap[s.studentId] || {};
    var base = PLAN_CONTENTS_[rs.nayami] || PLAN_CONTENTS_['_default'];
    var contents = base.concat(PLAN_LATE_AUG_);
    var lt = lastTimeMap[s.studentId];
    return {
      studentId: s.studentId,
      studentName: s.studentName,
      nayami: rs.nayami || '',
      perWeek: rs.r30 ? Math.max(1, Math.min(4, Math.round(rs.r30 / 4.3))) : 1,
      recentTime: lt ? planFmtTime_(lt.at) : null,
      contents: contents,
      existingPlans: existing[s.studentId] || []
    };
  });

  // 夏休みの終わりの目安（8/31）。9月以降に開いた場合は30日後
  var now2 = new Date();
  var defEnd = (now2.getMonth() <= 7)
    ? new Date(now2.getFullYear(), 7, 31)
    : new Date(now2.getTime() + 30 * 86400000);

  return {
    success: true,
    parentName: link.parentName || '',
    defaultEnd: planFmtDate_(defEnd),
    library: PLAN_CONTENT_LIBRARY_,
    students: students
  };
}

// plan_save：保存（所有検証つき）
// ========================================
function planSave(data) {
  var userId = data && data.userId;
  var studentId = data && data.studentId;
  var plans = (data && data.plans) || [];
  if (!userId || !studentId) return { success: false, code: 'INVALID_INPUT' };

  var link = planResolveParent_(userId);
  if (!link) return { success: false, code: 'NOT_LINKED' };
  var parentId = link.parentId;

  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // 所有検証：この生徒IDが本当にこの保護者の子か
  var stSh = ss.getSheetByName('students_lite');
  var sd = stSh.getDataRange().getValues();
  var sh0 = sd[0];
  var sSid = sh0.indexOf('生徒ID'), sPid = sh0.indexOf('保護者ID'), sNick = sh0.indexOf('ニックネーム');
  var studentName = null;
  for (var i = 1; i < sd.length; i++) {
    if (String(sd[i][sSid]) === String(studentId) && String(sd[i][sPid]) === String(parentId)) {
      studentName = sd[i][sNick] || '';
      break;
    }
  }
  if (studentName === null) return { success: false, code: 'NOT_FOUND' };

  ensurePlanSheets_();
  var pSh = ss.getSheetByName(PLAN_SHEET);
  var todayStr = planFmtDate_(new Date());

  // この生徒の「今日以降・状態=予定」の行をいったん消す（＝置きかえ）
  if (pSh.getLastRow() > 1) {
    var pd = pSh.getDataRange().getValues();
    var ph = pd[0];
    var cSid = ph.indexOf('生徒ID'), cDate = ph.indexOf('予定日'), cStat = ph.indexOf('状態');
    for (var r = pd.length - 1; r >= 1; r--) {
      if (String(pd[r][cSid]) !== String(studentId)) continue;
      if (String(pd[r][cStat]) !== '予定') continue;
      var ds = planToDateStr_(pd[r][cDate]);
      if (ds && ds >= todayStr) pSh.deleteRow(r + 1);
    }
  }

  // 追記
  var now = new Date();
  var rows = [];
  plans.forEach(function (p, idx) {
    if (!p || !p.date) return;
    if (String(p.date) < todayStr) return;   // 過去日は登録しない
    rows.push([
      'P' + now.getTime() + '_' + idx,
      parentId,
      studentId,
      studentName,
      p.date,
      p.time || '19:00',
      String(p.content || '').slice(0, 40),
      '予定',
      now,
      ''
    ]);
  });
  if (rows.length) {
    pSh.getRange(pSh.getLastRow() + 1, 1, rows.length, PLAN_HEADERS.length).setValues(rows);
  }

  // 埋まっている曜日を保存
  var cSh = ss.getSheetByName(PLAN_CONFIG_SHEET);
  var busy = (data.busyDays || []).join(',');
  var cd = cSh.getDataRange().getValues();
  var ch = cd[0];
  var kSid = ch.indexOf('生徒ID'), kBusy = ch.indexOf('埋まっている曜日'), kUpd = ch.indexOf('更新日時');
  var found = -1;
  for (var q = 1; q < cd.length; q++) {
    if (String(cd[q][kSid]) === String(studentId)) { found = q + 1; break; }
  }
  if (found > 0) {
    cSh.getRange(found, kBusy + 1).setValue(busy);
    cSh.getRange(found, kUpd + 1).setValue(now);
  } else {
    cSh.appendRow([parentId, studentId, busy, now]);
  }

  return { success: true, saved: rows.length };
}

// 前日リマインド（毎日10:00）
// ========================================
function sendPlanReminders() {
  var token = PropertiesService.getScriptProperties().getProperty('LINE_CHANNEL_ACCESS_TOKEN');
  if (!token) { console.error('LINE_CHANNEL_ACCESS_TOKEN 未設定'); return; }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var pSh = ss.getSheetByName(PLAN_SHEET);
  if (!pSh || pSh.getLastRow() < 2) { console.log('計画なし'); return; }

  var tomorrow = new Date();
  tomorrow.setDate(tomorrow.getDate() + 1);
  var tomorrowStr = planFmtDate_(tomorrow);

  // 保護者ID → userId
  var lk = ss.getSheetByName('紐付けマスタ').getDataRange().getValues();
  var lh = lk[0];
  var lPid = lh.indexOf('保護者ID'), lUid = lh.indexOf('LINE_userId'), lOpt = lh.indexOf('配信オプトイン');
  var uidMap = {};
  for (var i = 1; i < lk.length; i++) {
    if (lOpt >= 0 && lk[i][lOpt] !== true && String(lk[i][lOpt]).toUpperCase() !== 'TRUE') continue;
    uidMap[String(lk[i][lPid])] = lk[i][lUid];
  }

  // 「明日すでに確定予約が入っている生徒」には送らない（予約済みに催促しない）
  var alreadyBooked = {};
  var lsSh = ss.getSheetByName('lessons_lite');
  if (lsSh) {
    var ld = lsSh.getDataRange().getValues();
    var lh2 = ld[0];
    var xSid = lh2.indexOf('生徒ID'), xEnd = lh2.indexOf('授業の終了日時'), xConf = lh2.indexOf('確定');
    for (var j = 1; j < ld.length; j++) {
      var e = ld[j][xEnd];
      if (!e || typeof e.getMonth !== 'function') continue;
      var cf = String(ld[j][xConf]);
      if (cf !== '1' && cf !== '確定') continue;
      if (planFmtDate_(e) === tomorrowStr) alreadyBooked[String(ld[j][xSid])] = true;
    }
  }

  var pd = pSh.getDataRange().getValues();
  var ph = pd[0];
  var cPid = ph.indexOf('保護者ID'), cSid = ph.indexOf('生徒ID'), cName = ph.indexOf('生徒名'),
      cDate = ph.indexOf('予定日'), cTime = ph.indexOf('予定時刻'), cCont = ph.indexOf('やること'),
      cStat = ph.indexOf('状態'), cSent = ph.indexOf('リマインド送信日時');

  var sent = 0, skipped = 0;
  for (var r = 1; r < pd.length; r++) {
    if (String(pd[r][cStat]) !== '予定') continue;
    if (pd[r][cSent]) continue;
    if (planToDateStr_(pd[r][cDate]) !== tomorrowStr) continue;

    var sid = String(pd[r][cSid]);
    if (alreadyBooked[sid]) {
      pSh.getRange(r + 1, cSent + 1).setValue('予約済のためスキップ');
      skipped++; continue;
    }
    var uid = uidMap[String(pd[r][cPid])];
    if (!uid) { skipped++; continue; }

    var msg = buildPlanReminderFlex_({
      studentName: pd[r][cName] || 'お子さん',
      dateStr: tomorrowStr,
      time: planToTimeStr_(pd[r][cTime]),
      content: pd[r][cCont] || ''
    });

    var res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
      method: 'post', contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({ to: uid, messages: [msg] }),
      muteHttpExceptions: true
    });
    if (res.getResponseCode() === 200) {
      pSh.getRange(r + 1, cSent + 1).setValue(new Date());
      sent++;
    } else {
      console.error('push失敗', uid, res.getResponseCode(), res.getContentText());
    }
    Utilities.sleep(120);
  }
  console.log('前日リマインド 送信' + sent + '件 / スキップ' + skipped + '件（対象日 ' + tomorrowStr + '）');
}

    "contents": {
      "type": "bubble",

      "header": {
        "type": "box", "layout": "vertical",
        "backgroundColor": "#06C755", "paddingAll": "14px",
        "contents": [
          { "type": "text", "text": "明日の予定", "size": "xs", "color": "#E6F6EC" },
          { "type": "text", "text": dateLabel, "size": "lg", "weight": "bold", "color": "#FFFFFF", "wrap": true, "margin": "xs" }
        ]
      },
      "body": {
        "type": "box", "layout": "vertical", "paddingAll": "14px", "spacing": "sm",
        "contents": [
          {

// 当日朝リマインド（予約が入っている人向け・毎日7:30）
// ========================================
/**
 * 前日リマインド（計画があって予約がまだの人）とは別物。
 * こちらは「もう予約が入っている人」に、当日の朝「今日は授業の予定です」と伝える。
 *
 * キャンセルの扱い:
 *   lessons_lite の元クエリ Q330 は WHERE に canceled = 0 を持つ（2026-07-28 確認済み）。
 *   キャンセルされた授業は次の同期（15分毎）でシートから消えるので、ここには載らない。
 *   ただし confirmed は Q330 の WHERE に無いため、下で「確定」列を必ずチェックすること。
 *
 * 二重送信防止: 当日通知ログ シートに「日付＋保護者ID」で記録。1日1通まで。
 */
var PLAN_TODAY_LOG_SHEET = '当日通知ログ';
var PLAN_TODAY_LOG_HEADERS = ['日付', '保護者ID', '生徒名', '件数', '送信日時'];

function sendTodayLessonReminders() {
  var token = PropertiesService.getScriptProperties().getProperty('LINE_CHANNEL_ACCESS_TOKEN');
  if (!token) { console.error('LINE_CHANNEL_ACCESS_TOKEN 未設定'); return; }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var lsSh = ss.getSheetByName('lessons_lite');
  if (!lsSh || lsSh.getLastRow() < 2) { console.log('lessons_lite なし'); return; }

  var todayStr = planFmtDate_(new Date());

  // 1) 今日の確定授業を保護者ごとに集める
  var ld = lsSh.getDataRange().getValues();
  var h = ld[0];
  var cPid = h.indexOf('保護者ID'), cSid = h.indexOf('生徒ID'), cSName = h.indexOf('生徒のニックネーム'),
      cTName = h.indexOf('先生のニックネーム'), cEnd = h.indexOf('授業の終了日時'), cConf = h.indexOf('確定');

  var byParent = {};
  for (var i = 1; i < ld.length; i++) {
    var end = ld[i][cEnd];
    if (!end || typeof end.getMonth !== 'function') continue;
    if (planFmtDate_(end) !== todayStr) continue;
    var cf = String(ld[i][cConf]);
    if (cf !== '1' && cf !== '確定') continue;          // 未確定は送らない
    var pid = String(ld[i][cPid]);
    if (!byParent[pid]) byParent[pid] = [];
    byParent[pid].push({
      studentName: ld[i][cSName] || 'お子さん',
      teacherName: ld[i][cTName] || '',
      at: end
    });
  }
  var pids = Object.keys(byParent);
  if (!pids.length) { console.log('今日の確定授業なし'); return; }

  // 2) 紐付け＆オプトイン
  var lk = ss.getSheetByName('紐付けマスタ').getDataRange().getValues();
  var lh = lk[0];
  var lPid = lh.indexOf('保護者ID'), lUid = lh.indexOf('LINE_userId'), lOpt = lh.indexOf('配信オプトイン');
  var uidMap = {};
  for (var j = 1; j < lk.length; j++) {
    if (lOpt >= 0 && lk[j][lOpt] !== true && String(lk[j][lOpt]).toUpperCase() !== 'TRUE') continue;
    uidMap[String(lk[j][lPid])] = lk[j][lUid];
  }

  // 3) 送信済みログ
  var logSh = ss.getSheetByName(PLAN_TODAY_LOG_SHEET);
  if (!logSh) {
    logSh = ss.insertSheet(PLAN_TODAY_LOG_SHEET);
    logSh.getRange(1, 1, 1, PLAN_TODAY_LOG_HEADERS.length).setValues([PLAN_TODAY_LOG_HEADERS]).setFontWeight('bold');
    logSh.setFrozenRows(1);
  }
  var sentToday = {};
  if (logSh.getLastRow() > 1) {
    var gd = logSh.getDataRange().getValues();
    var gh = gd[0];
    var gDate = gh.indexOf('日付'), gPid = gh.indexOf('保護者ID');
    for (var k = 1; k < gd.length; k++) {
      if (planToDateStr_(gd[k][gDate]) === todayStr) sentToday[String(gd[k][gPid])] = true;
    }
  }

  // 4) 送信
  var sent = 0, skipped = 0, rows = [];
  pids.forEach(function (pid) {
    if (sentToday[pid]) { skipped++; return; }
    var uid = uidMap[pid];
    if (!uid) { skipped++; return; }

    var lessons = byParent[pid].sort(function (a, b) { return a.at - b.at; });
    var msg = buildTodayLessonFlex_(lessons);

    var res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
      method: 'post', contentType: 'application/json',
      headers: { Authorization: 'Bearer ' + token },
      payload: JSON.stringify({ to: uid, messages: [msg] }),
      muteHttpExceptions: true
    });
    if (res.getResponseCode() === 200) {
      rows.push([todayStr, pid, lessons[0].studentName, lessons.length, new Date()]);
      sent++;
    } else {
      console.error('push失敗', uid, res.getResponseCode(), res.getContentText());
    }
    Utilities.sleep(120);
  });
  if (rows.length) logSh.getRange(logSh.getLastRow() + 1, 1, rows.length, PLAN_TODAY_LOG_HEADERS.length).setValues(rows);
  console.log('当日リマインド 送信' + sent + '件 / スキップ' + skipped + '件（' + todayStr + '）');
}

// ユーティリティ
// ========================================
function planFmtDate_(d) {
  if (!d || typeof d.getMonth !== 'function') return '';
  var m = ('0' + (d.getMonth() + 1)).slice(-2);
  var dd = ('0' + d.getDate()).slice(-2);
  return d.getFullYear() + '-' + m + '-' + dd;
}
function planFmtTime_(d) {
  if (!d || typeof d.getMonth !== 'function') return null;
  return ('0' + d.getHours()).slice(-2) + ':' + ('0' + d.getMinutes()).slice(-2);
}
// シートの値が Date でも文字列でも 'YYYY-MM-DD' に揃える
function planToDateStr_(v) {
  if (!v) return '';
  if (typeof v.getMonth === 'function') return planFmtDate_(v);
  var s = String(v).trim().replace(/\//g, '-');
  var m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
  if (!m) return '';
  return m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2);
}
function planToTimeStr_(v) {
  if (!v) return '19:00';
  if (typeof v.getMonth === 'function') return planFmtTime_(v);
  var s = String(v).trim();
  var m = s.match(/(\d{1,2}):(\d{2})/);
  return m ? ('0' + m[1]).slice(-2) + ':' + m[2] : '19:00';
}
